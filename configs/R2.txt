! =====================================================
! R2 — LAN NEO NETWORK
! Role: ABR (Area Border Router) — OSPF Area 0 <-> Area 1
!       BDR for Area 0 (priority 100)
!       Router-on-a-Stick for VLAN 10 / VLAN 20 (Inter-VLAN routing)
!       Enforces ACL: ICMP range restriction (VLAN 10) + Telnet restriction (VLAN 20)
! =====================================================

version 15.1
no service timestamps log datetime msec
no service timestamps debug datetime msec
no service password-encryption
!
hostname R2
!
ip cef
no ipv6 cef
!
spanning-tree mode pvst
!
interface GigabitEthernet0/0
 ip address 10.0.0.2 255.255.255.0
 ip ospf priority 100
 duplex auto
 speed auto
!
interface GigabitEthernet0/1
 no ip address
 duplex auto
 speed auto
!
interface GigabitEthernet0/1.10
 encapsulation dot1Q 10
 ip address 10.2.10.1 255.255.255.0
 ip access-group BLOCK_PING_VLAN10 in
!
interface GigabitEthernet0/1.20
 encapsulation dot1Q 20
 ip address 10.2.20.1 255.255.255.0
 ip access-group BLOCK_TELNET_VLAN20 in
!
interface GigabitEthernet0/2
 no ip address
 duplex auto
 speed auto
 shutdown
!
interface Vlan1
 no ip address
 shutdown
!
router ospf 1
 router-id 2.2.2.2
 log-adjacency-changes
 network 10.0.0.0 0.0.0.255 area 0
 network 10.2.10.0 0.0.0.255 area 1
 network 10.2.20.0 0.0.0.255 area 1
!
ip classless
!
ip flow-export version 9
!
ip access-list extended BLOCK_PING_VLAN10
 deny icmp host 10.2.10.13 123.123.123.0 0.0.0.3 echo
 deny icmp 10.2.10.14 0.0.0.1 123.123.123.0 0.0.0.3 echo
 deny icmp 10.2.10.16 0.0.0.7 123.123.123.0 0.0.0.3 echo
 permit ip any any
!
ip access-list extended BLOCK_TELNET_VLAN20
 deny tcp host 10.2.20.100 host 123.123.123.2 eq telnet
 permit ip any any
!
line con 0
!
line aux 0
!
line vty 0 4
 login
!
end
