!       Static NAT (port forwarding): public 8.8.8.8:80/443 -> internal Web Server

version 15.1
no service timestamps log datetime msec
no service timestamps debug datetime msec
no service password-encryption
!
hostname R8
!
ip cef
no ipv6 cef
!
spanning-tree mode pvst
!
interface GigabitEthernet0/0
 ip address 8.8.8.1 255.255.255.240
 ip nat outside
 duplex auto
 speed auto
!
interface GigabitEthernet0/1
 ip address 192.168.100.1 255.255.255.0
 ip nat inside
 duplex auto
 speed auto
!
interface Vlan1
 no ip address
 shutdown
!
router eigrp 1000
 network 8.8.8.0 0.0.0.15
!
ip nat inside source list 1 interface GigabitEthernet0/0 overload
ip nat inside source static tcp 192.168.100.10 80 8.8.8.8 80
ip nat inside source static tcp 192.168.100.10 443 8.8.8.8 443
ip classless
!
ip flow-export version 9
!
access-list 1 permit 192.168.100.0 0.0.0.255
!
line con 0
!
line aux 0
!
line vty 0 4
 login
